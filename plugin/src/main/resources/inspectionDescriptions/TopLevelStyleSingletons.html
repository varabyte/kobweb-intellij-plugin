Reports Kobweb style-type properties (like CssStyle and Keyframes) that must be declared on the top level but aren't. Note that these singleton style properties can live inside top-level objects as well. 

// OK
val MyStyle = CssStyle { }
val MyKeyframes = Keyframes { }

// OK
object OrganizaingObject {
    val MyStyle = CssStyle { }
    val MyKeyframes = Keyframes { }
}

// Bad -- not top level
fun someFunction() {
    val MyStyle = CssStyle { }
}

// Bad -- not top level
class SomeClass {
    val MyKeyframes = Keyframes { }
}
