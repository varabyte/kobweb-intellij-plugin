Reports Kobweb properties (like val MyStyle = CssStyle) that must be declared on the top level but aren't. 

// OK
val MyStyle = CssStyle { }
val MyKeyframes = Keyframes { }

// OK
object OrganizaingObject {
    val MyStyle = CssStyle { }
    val MyKeyframes = Keyframes { }
}

// Bad -- not top level
fun someFunction() {
    val MyStyle = CssStyle { }
}

// Bad -- not top level
class SomeClass {
    val MyKeyframes = Keyframes { }
}
