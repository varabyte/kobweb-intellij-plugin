This action opts a page out of the default layout it uses (i.e. the one set by the closest @file:Layout annotation in its package), by annotating it with @NoLayout. 

// pages/blog/DefaultLayout.kt
@file:Layout(".components.layouts.BlogLayout")

// pages/blog/UndecoratedBlogPage.kt (before)
@Page
@Composable
fun UndecoratedBlogPage() { }

// pages/blog/UndecoratedBlogPage.kt (after)
@Page
@NoLayout
@Composable
fun UndecoratedBlogPage() { }
